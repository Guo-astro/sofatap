cmake_minimum_required(VERSION 3.20)
project(SofatapSecureInputProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(SofatapSecureInputProbeChecks ProbeChecks.cpp)
enable_testing()
add_test(NAME SecureInputProbeNoInputChecks COMMAND SofatapSecureInputProbeChecks)
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the diagnostic with 64-bit MSVC")
  endif()
  add_executable(SofatapSecureInputProbe Probe.cpp)
  set_target_properties(SofatapSecureInputProbe PROPERTIES OUTPUT_NAME "Sofatap.SecureInputProbe"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(SofatapSecureInputProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(SofatapSecureInputProbe PRIVATE /W4 /WX /sdl /guard:cf)
  target_link_options(SofatapSecureInputProbe PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /HIGHENTROPYVA)
  target_link_libraries(SofatapSecureInputProbe PRIVATE advapi32 user32 wtsapi32 shell32 ole32 uuid)
  target_compile_options(SofatapSecureInputProbeChecks PRIVATE /W4 /WX)
else()
  message(STATUS "Non-Windows: building no-input policy checks only")
endif()
