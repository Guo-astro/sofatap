cmake_minimum_required(VERSION 3.20)
project(SofatapLoginProvider LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The credential provider is Windows-only")
endif()
add_library(SofatapLoginProvider SHARED Provider.cpp Provider.def)
target_compile_features(SofatapLoginProvider PRIVATE cxx_std_20)
target_compile_definitions(SofatapLoginProvider PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(SofatapLoginProvider PRIVATE /W4 /guard:cf /sdl)
target_link_options(SofatapLoginProvider PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(SofatapLoginProvider PRIVATE advapi32 credui secur32 ole32 shlwapi wtsapi32 uuid netapi32)
add_executable(SofatapLoginProviderChecks ../Tests/ProviderChecks.cpp)
target_compile_features(SofatapLoginProviderChecks PRIVATE cxx_std_20)
target_compile_definitions(SofatapLoginProviderChecks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(SofatapLoginProviderChecks PRIVATE /W4 /sdl)
target_link_libraries(SofatapLoginProviderChecks PRIVATE advapi32 credui secur32 ole32 shlwapi wtsapi32 uuid netapi32)
enable_testing()
add_test(NAME ProviderChecks COMMAND SofatapLoginProviderChecks)
